{{- /* Renders the docs version of .page in the requested .format. Shared by the site-version shortcode (content) and the home layout (quick start). */ -}} {{- $format := .format | default "" -}} {{- $site := .page.Site -}} {{- $version := $site.Version.Name -}} {{- if eq $format "branch" -}} {{- if ne $version "main" -}} git checkout releases/v{{ $version }} {{ end -}} {{- else if eq $format "env" -}} {{- if ne $version "main" -}} OCM_VERSION={{ $version }} {{ end -}} {{- else if eq $format "constraint" -}} {{/* Version constraint for "ocm get cv --latest": the newest stable patch of this docs version (patch releases are not tracked here, and X.Y.0 is not always published), or the newest stable release on "main". Mirrors what install-cli.sh installs for the "env" format. */}} {{- if eq $version "main" -}} >0.0.0 {{- else -}} ~{{ $version }} {{- end -}} {{- else if eq $format "semver" -}} {{/* Full major.minor.0 for OCM component/image references. The version config only carries major.minor names, so .0 is appended (patches are not tracked here). On the "main" docs there is no installable "main" component, so the reference resolves to the latest release: the default content version. */}} {{- $base := $version -}} {{- if eq $version "main" -}} {{- range $site.Home.Rotate "version" -}} {{- if .Site.Version.IsDefault -}} {{- $base = .Site.Version.Name -}} {{- end -}} {{- end -}} {{- end -}} {{- $base -}}.0 {{- else -}} {{- $version -}} {{- end -}}
